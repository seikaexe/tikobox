Di halaman ini

  • 2. Unduh File
  • Selanjutnya

Alur dasar ada dua langkah: ambil info lewat /api/tiktok, lalu unduh file lewat /api/download

1. Ambil Info Link#

Kirim link TikTok ke parameter url. Link boleh berupa link pendek (vt.tiktok.com, vm.tiktok.com, tt.site) maupun link lengkap

cURL
curl -sS -G "https://tikobox.my.id/api/tiktok" \
  -A "MyTikoboxClient/1.0" \
  --data-urlencode "url=https://vt.tiktok.com/ZSxxxxxxx/"
Fetch
const url = new URL("https://tikobox.my.id/api/tiktok")
url.searchParams.set("url", "https://vt.tiktok.com/ZSxxxxxxx/")

const res = await fetch(url)
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json.data)
Node.js
const url = new URL("https://tikobox.my.id/api/tiktok")
url.searchParams.set("url", "https://vt.tiktok.com/ZSxxxxxxx/")

const res = await fetch(url, { headers: { "User-Agent": "MyTikoboxClient/1.0" } })
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json.data)
Python
import requests

res = requests.get(
    "https://tikobox.my.id/api/tiktok",
    params={
        "url": "https://vt.tiktok.com/ZSxxxxxxx/",
    },
    headers={"User-Agent": "MyTikoboxClient/1.0"},
    timeout=35,
)

data = res.json()

if data["code"] != 0:
    raise RuntimeError(f'{res.status_code} {data["msg"]}')

print(data["data"])
PHP
<?php

$query = http_build_query([
    "url" => "https://vt.tiktok.com/ZSxxxxxxx/",
]);

$ch = curl_init("https://tikobox.my.id/api/tiktok" . "?" . $query);
curl_setopt_array($ch, [
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_TIMEOUT => 35,
    CURLOPT_USERAGENT => "MyTikoboxClient/1.0",
]);

$body = curl_exec($ch);
if ($body === false) {
    throw new RuntimeException(curl_error($ch));
}

$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
$json = json_decode($body, true);

if (($json["code"] ?? -1) !== 0) {
    throw new RuntimeException($status . " " . ($json["msg"] ?? "error"));
}

print_r($json["data"]);

Cek code pada body. Nilai 0 berarti sukses dan data ada di data. Nilai -1 berarti gagal dan alasannya ada di msg

Response
{
  "code": 0,
  "msg": "success",
  "processed_time": 1.2345,
  "data": {
    "id": "7300000000000000000",
    "title": "Contoh judul video",
    "play": "https://v16m.tiktokcdn.com/xxxx/video.mp4",
    "music": "https://sf16.tiktokcdn.com/xxxx/music.mp3",
    "...": "lihat Format Response"
  }
}

2. Unduh File#

Pakai URL dari data.play, data.music, atau data.images sebagai parameter url di /api/download

cURL
curl -sS -f -G "https://tikobox.my.id/api/download" \
  -A "MyTikoboxClient/1.0" \
  --data-urlencode "url=https://v16m.tiktokcdn.com/xxxx/video.mp4" \
  --data-urlencode "name=video.mp4" \
  -o "video.mp4"
Fetch
const url = new URL("https://tikobox.my.id/api/download")
url.searchParams.set("url", "https://v16m.tiktokcdn.com/xxxx/video.mp4")
url.searchParams.set("name", "video.mp4")

const res = await fetch(url)
if (!res.ok) throw new Error((await res.json()).msg)

const blob = await res.blob()
console.log(blob.type, blob.size)
Node.js
import { createWriteStream } from "node:fs"
import { Readable } from "node:stream"
import { pipeline } from "node:stream/promises"

const url = new URL("https://tikobox.my.id/api/download")
url.searchParams.set("url", "https://v16m.tiktokcdn.com/xxxx/video.mp4")
url.searchParams.set("name", "video.mp4")

const res = await fetch(url, { headers: { "User-Agent": "MyTikoboxClient/1.0" } })
if (!res.ok) throw new Error((await res.json()).msg)

await pipeline(Readable.fromWeb(res.body), createWriteStream("video.mp4"))
console.log("tersimpan:", "video.mp4")
Python
import requests

res = requests.get(
    "https://tikobox.my.id/api/download",
    params={
        "url": "https://v16m.tiktokcdn.com/xxxx/video.mp4",
        "name": "video.mp4",
    },
    headers={"User-Agent": "MyTikoboxClient/1.0"},
    stream=True,
    timeout=35,
)

if not res.ok:
    raise RuntimeError(res.json()["msg"])

with open("video.mp4", "wb") as f:
    for chunk in res.iter_content(chunk_size=65536):
        f.write(chunk)
PHP
<?php

$query = http_build_query([
    "url" => "https://v16m.tiktokcdn.com/xxxx/video.mp4",
    "name" => "video.mp4",
]);

$ch = curl_init("https://tikobox.my.id/api/download" . "?" . $query);
$file = fopen("video.mp4", "wb");

curl_setopt_array($ch, [
    CURLOPT_FILE => $file,
    CURLOPT_FOLLOWLOCATION => false,
    CURLOPT_TIMEOUT => 0,
    CURLOPT_USERAGENT => "MyTikoboxClient/1.0",
]);

if (!curl_exec($ch)) {
    throw new RuntimeException(curl_error($ch));
}

$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
fclose($file);

if ($status !== 200) {
    unlink("video.mp4");
    throw new RuntimeException("HTTP $status");
}
URL media dari TikTok bisa kedaluwarsa. Unduh segera setelah mengambil info, jangan disimpan untuk dipakai berhari-hari

Selanjutnya#

  • Contoh klien lengkap: cURL, JavaScript, Node.js, Python, PHP
  • Semua field response ada di Format Response
