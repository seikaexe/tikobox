GET/api/

Tanpa parameter, mengembalikan ringkasan penggunaan. Dengan ?url=LINK, endpoint ini langsung memproses link seperti /api/tiktok. Termasuk dalam kelompok rate limit lookup

Contoh Request#

cURL
curl -sS "https://tikobox.my.id/api/" \
  -A "MyTikoboxClient/1.0"
Fetch
const url = new URL("https://tikobox.my.id/api/")

const res = await fetch(url)
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json)
Node.js
const url = new URL("https://tikobox.my.id/api/")

const res = await fetch(url, { headers: { "User-Agent": "MyTikoboxClient/1.0" } })
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json)
Python
import requests

res = requests.get(
    "https://tikobox.my.id/api/",
    headers={"User-Agent": "MyTikoboxClient/1.0"},
    timeout=35,
)

data = res.json()

if data["code"] != 0:
    raise RuntimeError(f'{res.status_code} {data["msg"]}')

print(data)
PHP
<?php

$ch = curl_init("https://tikobox.my.id/api/");
curl_setopt_array($ch, [
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_TIMEOUT => 35,
    CURLOPT_USERAGENT => "MyTikoboxClient/1.0",
]);

$body = curl_exec($ch);
if ($body === false) {
    throw new RuntimeException(curl_error($ch));
}

$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
$json = json_decode($body, true);

if (($json["code"] ?? -1) !== 0) {
    throw new RuntimeException($status . " " . ($json["msg"] ?? "error"));
}

print_r($json);
