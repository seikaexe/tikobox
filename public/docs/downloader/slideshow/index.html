Slideshow Downloader

Ambil semua foto dari postingan foto TikTok

Postingan foto (slideshow) memakai endpoint /api/tiktok. Bedanya, response memiliki field images dan tidak memiliki video

Mendeteksi Postingan Foto#

  • data.images hanya ada untuk postingan foto, berupa array URL gambar
  • data.play, data.wmplay, dan data.hdplay berisi string kosong
  • data.size, data.wm_size, dan data.hd_size bernilai 0
  • data.music tetap berisi audio bila tersedia

Contoh Request#

cURL
curl -sS -G "https://tikobox.my.id/api/tiktok" \
  -A "MyTikoboxClient/1.0" \
  --data-urlencode "url=https://www.tiktok.com/@username/photo/7300000000000000000"
Fetch
const url = new URL("https://tikobox.my.id/api/tiktok")
url.searchParams.set("url", "https://www.tiktok.com/@username/photo/7300000000000000000")

const res = await fetch(url)
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json.data)
Node.js
const url = new URL("https://tikobox.my.id/api/tiktok")
url.searchParams.set("url", "https://www.tiktok.com/@username/photo/7300000000000000000")

const res = await fetch(url, { headers: { "User-Agent": "MyTikoboxClient/1.0" } })
const json = await res.json()

if (json.code !== 0) throw new Error(`${res.status} ${json.msg}`)
console.log(json.data)
Python
import requests

res = requests.get(
    "https://tikobox.my.id/api/tiktok",
    params={
        "url": "https://www.tiktok.com/@username/photo/7300000000000000000",
    },
    headers={"User-Agent": "MyTikoboxClient/1.0"},
    timeout=35,
)

data = res.json()

if data["code"] != 0:
    raise RuntimeError(f'{res.status_code} {data["msg"]}')

print(data["data"])
PHP
<?php

$query = http_build_query([
    "url" => "https://www.tiktok.com/@username/photo/7300000000000000000",
]);

$ch = curl_init("https://tikobox.my.id/api/tiktok" . "?" . $query);
curl_setopt_array($ch, [
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_TIMEOUT => 35,
    CURLOPT_USERAGENT => "MyTikoboxClient/1.0",
]);

$body = curl_exec($ch);
if ($body === false) {
    throw new RuntimeException(curl_error($ch));
}

$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
$json = json_decode($body, true);

if (($json["code"] ?? -1) !== 0) {
    throw new RuntimeException($status . " " . ($json["msg"] ?? "error"));
}

print_r($json["data"]);

Mengunduh Foto#

Unduh setiap URL pada images lewat /api/download. Tipe gambar yang diizinkan: JPEG, PNG, WebP, GIF, AVIF, HEIC

cURL
curl -sS -f -G "https://tikobox.my.id/api/download" \
  -A "MyTikoboxClient/1.0" \
  --data-urlencode "url=https://p16-sign.tiktokcdn.com/xxxx/foto-1.jpeg" \
  --data-urlencode "name=foto-1.jpg" \
  -o "foto-1.jpg"
Fetch
const url = new URL("https://tikobox.my.id/api/download")
url.searchParams.set("url", "https://p16-sign.tiktokcdn.com/xxxx/foto-1.jpeg")
url.searchParams.set("name", "foto-1.jpg")

const res = await fetch(url)
if (!res.ok) throw new Error((await res.json()).msg)

const blob = await res.blob()
console.log(blob.type, blob.size)
Node.js
import { createWriteStream } from "node:fs"
import { Readable } from "node:stream"
import { pipeline } from "node:stream/promises"

const url = new URL("https://tikobox.my.id/api/download")
url.searchParams.set("url", "https://p16-sign.tiktokcdn.com/xxxx/foto-1.jpeg")
url.searchParams.set("name", "foto-1.jpg")

const res = await fetch(url, { headers: { "User-Agent": "MyTikoboxClient/1.0" } })
if (!res.ok) throw new Error((await res.json()).msg)

await pipeline(Readable.fromWeb(res.body), createWriteStream("foto-1.jpg"))
console.log("tersimpan:", "foto-1.jpg")
Python
import requests

res = requests.get(
    "https://tikobox.my.id/api/download",
    params={
        "url": "https://p16-sign.tiktokcdn.com/xxxx/foto-1.jpeg",
        "name": "foto-1.jpg",
    },
    headers={"User-Agent": "MyTikoboxClient/1.0"},
    stream=True,
    timeout=35,
)

if not res.ok:
    raise RuntimeError(res.json()["msg"])

with open("foto-1.jpg", "wb") as f:
    for chunk in res.iter_content(chunk_size=65536):
        f.write(chunk)
PHP
<?php

$query = http_build_query([
    "url" => "https://p16-sign.tiktokcdn.com/xxxx/foto-1.jpeg",
    "name" => "foto-1.jpg",
]);

$ch = curl_init("https://tikobox.my.id/api/download" . "?" . $query);
$file = fopen("foto-1.jpg", "wb");

curl_setopt_array($ch, [
    CURLOPT_FILE => $file,
    CURLOPT_FOLLOWLOCATION => false,
    CURLOPT_TIMEOUT => 0,
    CURLOPT_USERAGENT => "MyTikoboxClient/1.0",
]);

if (!curl_exec($ch)) {
    throw new RuntimeException(curl_error($ch));
}

$status = curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
fclose($file);

if ($status !== 200) {
    unlink("foto-1.jpg");
    throw new RuntimeException("HTTP $status");
}
